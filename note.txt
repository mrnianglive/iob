//View operations Query 
CREATE VIEW operations AS ( SELECT TbleOperations.RefOperations,TbleAgency.RefAgency,TbleAgency.NameAgency,TbleProduit.RefProduit,TbleProduit.NameProduit,TbleType.RefType,TbleType.NameType,TbleOperations.NameClient, TbleOperations.NumCompte,TbleOperations.MontantVersement,TbleOperations.Remarque,TbleOperations.Approve2_Time,TbleUsers.login,TbleOperations.SentFromAgency, TbleOperations.datePayement,TbleOperations.Approve2_Id,TbleOperations.Reset_Id,TbleOperations.Approve1_Time,TbleOperations.Bordereau,TbleOperations.Reset_At,TbleOperations.DateValidate,TbleOperations.RefValidate,TbleOperations.TypeAppro,TbleOperations.TypeRetrait,TbleOperations.uniqid,TbleOperations.ValidateDate,TbleOperations.Validate,TbleOperations.RefCaisse,TbleCaisse.NameCaisse,TbleOperations.Insert_Time FROM TbleOperations INNER JOIN TbleType ON TbleType.RefType=TbleOperations.RefType INNER JOIN TbleCaisse ON TbleCaisse.RefCaisse=TbleOperations.RefCaisse INNER JOIN TbleAgency ON TbleAgency.RefAgency=TbleCaisse.RefAgency LEFT JOIN TbleProduit ON TbleProduit.RefProduit=TbleOperations.RefProduit INNER JOIN TbleUsers ON TbleUsers.Refusers=TbleOperations.Insert_Id)
